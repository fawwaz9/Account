# What this app actually imports, rather than everything that was ever in the
# virtualenv. The previous file was a raw `pip freeze`, so it carried openai,
# firebase_admin, three google-cloud packages and grpcio -- none of which appear
# in a single import -- and every transitive dependency of those, pinned as
# though the app depended on them directly.
#
# Only direct dependencies are listed. pip resolves the rest, which is what
# keeps this readable: `starlette` is here because fastapi needs it, not because
# this project chose a version of it.

# ---- Web ----
fastapi==0.135.1
# The [standard] extras are what `--reload` and the production server use:
# watchfiles for reloading, uvloop and httptools for speed.
uvicorn[standard]==0.41.0

# ---- Data ----
SQLAlchemy==2.0.51
# The Postgres driver. Version 3, not psycopg2: it is the current generation and
# first-class in SQLAlchemy 2.0, where psycopg2 is in maintenance. The [binary]
# extra ships prebuilt wheels, so the Docker build needs no compiler.
# Not asyncpg -- that needs an async engine, and every service here is sync.
psycopg[binary]==3.3.5
pydantic==2.12.5
pydantic-settings==2.14.2

# ---- Auth ----
bcrypt==5.0.0
PyJWT==2.11.0
# `google.auth.transport.requests` is the transport this app uses to verify a
# Google ID token, and it needs `requests` present -- hence the extra rather
# than a bare google-auth.
google-auth[requests]==2.49.0

# ---- Tests ----
# Previously undeclared: the suite has always needed it, and a fresh clone
# following the README could not run a single test.
pytest==9.1.1
# Starlette's TestClient is built on httpx and three test files use it, but
# httpx ships only under fastapi[standard] / starlette[full] and this pins
# plain fastapi -- so it was installed here by accident and absent from a
# fresh clone. Same gap as pytest above, found the same way.
httpx==0.28.1
